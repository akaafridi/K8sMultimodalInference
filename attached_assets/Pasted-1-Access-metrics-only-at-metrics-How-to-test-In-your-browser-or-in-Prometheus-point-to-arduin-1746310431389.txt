2. Give users a proper “home page” for your Web UI
Since you removed the default Replit stub, Flask has no “/” route to serve your Bootstrap UI. Let’s add one.

Create a simple HTML template

File: templates/index.html

html
Copy
Edit
<!doctype html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width,initial-scale=1">
  <title>BLIP Image Captioning</title>
  <link 
    href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/css/bootstrap.min.css" 
    rel="stylesheet">
</head>
<body class="p-5">
  <div class="container">
    <h1 class="mb-4">BLIP Image Captioning</h1>
    <form id="infer-form">
      <div class="mb-3">
        <label for="imageInput" class="form-label">Image URL or Base64</label>
        <input type="text" class="form-control" id="imageInput" placeholder="Enter image URL or paste base64">
      </div>
      <button type="submit" class="btn btn-primary">Generate Caption</button>
    </form>
    <pre id="output" class="mt-4 bg-light p-3"></pre>
  </div>
  <script>
    document.getElementById('infer-form').onsubmit = async e => {
      e.preventDefault();
      const data = { image: document.getElementById('imageInput').value };
      const res  = await fetch('/infer', {
        method: 'POST',
        headers: {'Content-Type':'application/json'},
        body: JSON.stringify(data)
      });
      const json = await res.json();
      document.getElementById('output').textContent = JSON.stringify(json, null, 2);
    };
  </script>
</body>
</html>
