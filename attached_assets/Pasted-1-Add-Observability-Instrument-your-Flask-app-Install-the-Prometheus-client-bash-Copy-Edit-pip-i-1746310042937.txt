1. Add Observability
Instrument your Flask app

Install the Prometheus client:

Copy
Edit

In model_server.py, add:

python
Copy
Edit
from prometheus_client import Counter, Histogram, generate_latest, CONTENT_TYPE_LATEST

# Metrics
INFER_COUNT = Counter("infer_requests_total", "Total inference requests")
INFER_LATENCY = Histogram("infer_latency_seconds", "Inference latency")

@app.route("/metrics")
def metrics():
    return generate_latest(), 200, {"Content-Type": CONTENT_TYPE_LATEST}

@app.before_request
def before():
    request.start_time = time.time()

@app.after_request
def after(response):
    if request.endpoint == "infer":
        INFER_COUNT.inc()
        INFER_LATENCY.observe(time.time() - request.start_time)
    return response

Build dashboards & alerts

In Grafana, import a basic template for Flask apps or create panels for:

infer_latency_seconds (histogram percentiles)

GPU node metrics (e.g., nvidia_smi_utilization_gpu)

Define an alert rule: e.g.

yaml
Copy
Edit
- alert: HighInferLatency
  expr: histogram_quantile(0.95, sum(rate(infer_latency_seconds_bucket[5m])) by (le)) > 1
  for: 2m
  labels:
    severity: warning
  annotations:
    summary: "P95 inference latency > 1s"

from transformers import BlipProcessor, BlipForConditionalGeneration
import torch

# load once
processor = BlipProcessor.from_pretrained("Salesforce/blip-image-captioning-base")
model     = BlipForConditionalGeneration.from_pretrained("Salesforce/blip-image-captioning-base").to("cuda")

@app.route("/infer", methods=["POST"])
def infer():
    INFER_COUNT.inc()
    start = time.time()

    payload = request.json
    image   = decode_image(payload["image"])  # implement decode_image

    inputs  = processor(image, return_tensors="pt").to("cuda")
    out     = model.generate(**inputs)
    caption = processor.decode(out[0], skip_special_tokens=True)

    INFER_LATENCY.observe(time.time() - start)
    return jsonify({
      "predictions":   caption,
      "model_version": model.config._name_or_path
    })
Test locally with Docker (or your GPU‑enabled dev environment) to verify you get real captions.
